CREATE TABLE Employees (
    Emp_ID INT PRIMARY KEY,

    Dept_ID INT,
    Salary DECIMAL(10,2),

##Find employees whose salary is greater than 50000.
select *
from employees
where salary > 50000;


##Find employees working in the IT department.
select*
from employees
where  department ="IT";

##Find employees whose salary is less than or equal to 45000.
select *
from employees
where salary <= 45000;


##Find all employees who belong to the IT department AND have a salary greater than 50000.

select *
from employees
where Department = 'IT' AND salary > 50000;

select *
from employees
where Department = 'IT' OR Department = 'Finance';

##Find all employees who belong to departments with Dept_ID 1, 2, or 3.
select *
from employees
where Dept_ID in (1,2,3);


##Find all employees whose salary is between 40,000 and 60,000 (including both values).
select *
from employees
where salary between 40000 AND 60000;

##Find all employees whose names start with the letter 'A'.
select*
from employees
where name like 'A%';

##Find all unique cities from the Employees table.

from Employees;

##Display all employees and arrange their salaries from highest to lowest.
select *
from employees

##Find the top 5 highest-paid employees from the Employees table.
select *
from employees

##Find the total number of employees in each department using the Employees table.
select dept_id, count(dept_id)
from employees
group by dept_id;

select dept_id,avg(salary)
from employees
group by Dept_id;


##Find the total salary paid in each city.
select city, sum("Salary") as total_salary
from employees
group by city;

##Find the number of employees in each city, but display only those cities where employee count is greater than 2.
select city, count(emp_id)
from employees
group by city
Having count(emp_id) > 2;

##Find departments where the average salary is greater than 50000.
select dept_id,avg(salary)
from employees
group by dept_id
having avg(salary) > 50000;


##Find all employees whose salary is greater than the average salary of all employees.
select *
from employees

##Find all employees who earn the highest salary in the company.
